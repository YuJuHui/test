git add 想保存的文件名     //保存文件命令， .表示保存所有内容
git commit -m '这次更改的描述'      //添加描述信息
git checkout -m 哈希码      //回到hash码指示的时间位置
git branch -M main      //选择项目分支

git remote add origin 项目地址   //选择项目地址

 
git pull origin main --allow-unrelated-histories   //先拉取远程更改并合并
